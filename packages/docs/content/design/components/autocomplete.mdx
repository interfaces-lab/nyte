```ts title="Import"
  AutocompleteGroupLabel,

  AutocompleteSeparator,
  AutocompleteStatus,
  AutocompleteTrigger,
  AutocompleteValue,
} from "@nyte-ai/ui/autocomplete";
```

A search field for a long list, such as fonts or models. For a short list, use
[Select](/design/select).

## Usage

- Put the current value or `AutocompleteValue` inside `AutocompleteTrigger`.
- Pass the data to `items` on `Autocomplete` and render each entry from a function child of
  `AutocompleteList`. Inside an `AutocompleteGroup`, render its entries with
  `AutocompleteCollection`.
- Give each `AutocompleteItem` a `value`.
- `value` and `onValueChange` hold the selection. Clearing passes `null` in single-selection mode.
  With `multiple`, values and callbacks use arrays.
- `inputValue` and `onInputValueChange` hold the search text.
- To nest the list in another surface, such as a submenu, set `inline` and `open` on
  `Autocomplete`, drop `AutocompleteContent`, and set [`variant="inline"`](#variant).
- Keep `AutocompleteStatus` and `AutocompleteEmpty` mounted and change their children, so screen
  readers announce the change.

## Anatomy

```tsx
<Autocomplete>
  <AutocompleteTrigger aria-label="Model">
    <AutocompleteValue />
  </AutocompleteTrigger>
  <AutocompleteContent>
    <AutocompleteInputGroup>
      <AutocompleteInput aria-label="Search models" />
      <AutocompleteClear aria-label="Clear model">Clear</AutocompleteClear>
    </AutocompleteInputGroup>
    <AutocompleteStatus />
    <AutocompleteList>
      <AutocompleteGroup>
        <AutocompleteSeparator />
        <AutocompleteGroupLabel />
        <AutocompleteCollection>
          {(item: string) => <AutocompleteItem value={item}>{item}</AutocompleteItem>}
        </AutocompleteCollection>
      </AutocompleteGroup>
    </AutocompleteList>
    <AutocompleteEmpty />
  </AutocompleteContent>
</Autocomplete>
```

## Examples

### Searchable picker

<Preview>
  <AutocompleteDemo />
</Preview>

```tsx
interface ModelGroup {
  readonly label: string;
  readonly items: readonly string[];
}

const groups: readonly ModelGroup[] = [
  { label: "Anthropic", items: ["claude-opus-4-6", "claude-sonnet-4-6", "claude-haiku-4-5"] },
  { label: "OpenAI", items: ["{{modelId}}"] },
  { label: "Google", items: ["gemini-3-pro"] },
];

const [model, setModel] = useState("claude-sonnet-4-6");
const [search, setSearch] = useState("");

<Autocomplete
  items={groups}
  value={model}
  onValueChange={(next) => {
    if (next !== null) setModel(next);
  }}
  inputValue={search}
  onInputValueChange={setSearch}
  autoHighlight
>
  <AutocompleteTrigger aria-label="Model">{model}</AutocompleteTrigger>
  <AutocompleteContent>
    <AutocompleteInput aria-label="Search models" placeholder="Search models" />
    <AutocompleteList>
      {(group: ModelGroup, index: number) => (
        <AutocompleteGroup key={group.label} items={group.items}>
          {index > 0 && <AutocompleteSeparator />}
          <AutocompleteGroupLabel>{group.label}</AutocompleteGroupLabel>
          <AutocompleteCollection>
            {(item: string) => (
              <AutocompleteItem key={item} value={item}>
                {item}
              </AutocompleteItem>
            )}
          </AutocompleteCollection>
        </AutocompleteGroup>
      )}
    </AutocompleteList>
    <AutocompleteEmpty>No models match</AutocompleteEmpty>
  </AutocompleteContent>
</Autocomplete>;
```

### Inside a submenu

The [Menu](/design/menu) submenu supplies the popup. `filter={null}` leaves filtering to
you.

```tsx
<MenuSubContent>
  <Autocomplete
    inline
    open
    filter={null}
    autoHighlight
    items={options}
    value={current}
    onValueChange={(next) => {
      if (next !== null) choose(next);
    }}
    inputValue={search}
    onInputValueChange={setSearch}
  >
    <AutocompleteInput variant="inline" aria-label="Search models" placeholder="Search models" />
    <MenuSeparator />
    <AutocompleteList variant="inline">
      {(option: ModelOption) => (
        <AutocompleteItem variant="inline" key={option.key} value={option}>
          {option.name}
        </AutocompleteItem>
      )}
    </AutocompleteList>
  </Autocomplete>
</MenuSubContent>
```

## Accessibility

- Name the input and trigger with `aria-label`. The trigger text is only the current value.
- Arrow keys move the highlight while focus stays in the input. Enter picks. Escape closes.
