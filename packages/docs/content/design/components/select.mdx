```ts title="Import"
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@nyte-ai/ui/select";
```

For a searchable list, use [Autocomplete](/design/autocomplete). For a list of
actions, use [Menu](/design/menu).

## Usage

- Pass `items` to `Select` so `SelectValue` shows the chosen label instead of the raw value.
- `onValueChange` can receive `null` in single-selection mode. With `multiple`, values and callbacks use arrays.

## Anatomy

```tsx
<Select>
  <SelectTrigger>
    <SelectValue />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="choice">Choice</SelectItem>
  </SelectContent>
</Select>
```

## Examples

</Preview>

```tsx
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@nyte-ai/ui/select";
import { useState } from "react";

type Theme = "system" | "light" | "dark";

const themes = [
  { value: "system", label: "System" },
  { value: "light", label: "Light" },
  { value: "dark", label: "Dark" },
] as const satisfies readonly { readonly value: Theme; readonly label: string }[];

function ThemeSelect() {
  const [theme, setTheme] = useState<Theme>("system");

  return (
    <Select
      items={themes}
      value={theme}
      onValueChange={(next) => {
        if (next !== null) setTheme(next);
      }}
    >
      <SelectTrigger aria-label="Theme">
        <SelectValue />
      </SelectTrigger>
      <SelectContent>
        {themes.map((option) => (
          <SelectItem key={option.value} value={option.value} label={option.label}>
            {option.label}
          </SelectItem>
        ))}
      </SelectContent>
    </Select>
  );
}
```

```tsx
<SelectTrigger aria-label="Default model" width="wide">
  <SelectValue />
</SelectTrigger>
```

### SelectContent

The popup is the listbox. `render`, `className`, `style`, and `xstyle` target the popup.

| Prop                   | Type                                                                       | Default                                                      | Description                                    |
| ---------------------- | -------------------------------------------------------------------------- | ------------------------------------------------------------ | ---------------------------------------------- |
| `side`                 | `"top" \| "bottom" \| "left" \| "right" \| "inline-start" \| "inline-end"` | `"bottom"`                                                   |                                                |
| `align`                | `"start" \| "center" \| "end"`                                             | `"end"`                                                      |                                                |
| `sideOffset`           | `number \| OffsetFunction`                                                 | `4`                                                          | Ignored during trigger alignment.              |
| `alignOffset`          | `number \| OffsetFunction`                                                 | `0`                                                          |                                                |
| `alignItemWithTrigger` | `boolean`                                                                  | `true`                                                       | Aligns selection over trigger for mouse input. |
| `collisionAvoidance`   | `CollisionAvoidance`                                                       | `{ side: "flip", align: "shift", fallbackAxisSide: "none" }` |                                                |
| `tint`                 | `Tint`                                                                     |                                                              |                                                |
| `xstyle`               | `Restyle<"SelectContent">`                                                 |                                                              |                                                |
